                } else {
                    // 실제 거리에서 **절반까지만** 깎아 어림값을 만든다.
                    //
                    // h 는 실제보다 크면 안 된다(그러면 A* 가 최단 경로를 놓친다).
                    // 예전에는 실제 거리에 0.2~0.7 배를 곱해 깎았는데, 그러면 거리가 2 든 3 이든
                    // 반올림 뒤 바닥값에 걸려 **전부 1 이 되었다.** 화면에는 「목표까지 남은
                    // 거리의 추정값」이라 적어 두고 값은 거리에 따라 변하지 않는 셈이라,
                    // 학생이 h 를 거리와 무관한 무엇으로 읽게 된다.
                    // 깎는 폭을 절반으로 묶으면 h 가 거리를 따라가면서도 실제를 넘지 않는다.
                    const d = treeDist(id);
                    h = d - Math.floor(Math.random() * (Math.floor(d / 2) + 1));
                }
